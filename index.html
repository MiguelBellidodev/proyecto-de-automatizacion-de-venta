Arquitectura de Base de Datos (PostgreSQL)

Esquema relacional con llaves foráneas, índices de alto rendimiento y trazabilidad de eventos.

-- Schema: Automatización de Ventas
-- Autor: Mateo (Backend & DB Engineer)

CREATE TABLE IF NOT EXISTS leads (
  id SERIAL PRIMARY KEY,
  nombre VARCHAR(255) NOT NULL,
  email VARCHAR(255),
  telefono VARCHAR(50) NOT NULL,
  empresa VARCHAR(255),
  score INT DEFAULT 0,
  etapa VARCHAR(50) DEFAULT 'nuevo',
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS deals (
  id SERIAL PRIMARY KEY,
  lead_id INT REFERENCES leads(id) ON DELETE CASCADE,
  titulo VARCHAR(255) NOT NULL,
  valor NUMERIC(14,2) NOT NULL DEFAULT 0,
  probabilidad INT DEFAULT 50,
  estado VARCHAR(50) DEFAULT 'abierto',
  creado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS mensajes_enviados (
  id SERIAL PRIMARY KEY,
  lead_id INT REFERENCES leads(id),
  canal VARCHAR(50) DEFAULT 'whatsapp',
  contenido TEXT NOT NULL,
  estado VARCHAR(50) DEFAULT 'entregado',
  enviado_en TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_leads_score ON leads(score DESC);
CREATE INDEX idx_deals_estado ON deals(estado);
